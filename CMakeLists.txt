cmake_minimum_required(VERSION 3.20)
project(AeroRAM VERSION 0.4.3 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "AeroRAM requires a Windows target toolchain.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build the Windows x64 target.")
endif()
add_executable(AeroRAM WIN32 src/app.cpp src/memory.cpp src/system_info.cpp resources/app.rc)
target_compile_features(AeroRAM PRIVATE cxx_std_17)
target_compile_definitions(AeroRAM PRIVATE UNICODE _UNICODE)
target_include_directories(AeroRAM PRIVATE resources)
target_link_libraries(AeroRAM PRIVATE gdiplus dwmapi shell32 ole32 oleaut32 wbemuuid dxguid dxgi d3d11 uuid psapi advapi32 user32 gdi32)
if(MSVC)
  target_compile_options(AeroRAM PRIVATE /W4 /utf-8)
  set_property(TARGET AeroRAM PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(AeroRAM PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(AeroRAM PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
